3. Singleton de Conexión (ConexionBD.java)
Garantiza el punto centralizado de acceso al motor de base de datos
.
package edu.usal.util;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;

public class ConexionBD {
    private static ConexionBD instancia;
    private static final String URL = "jdbc:mysql://localhost:3306/mi_base_datos?useSSL=false&serverTimezone=UTC";
    private static final String USER = "root";
    private static final String PASS = "root";

    private ConexionBD() {
        try {
            Class.forName("com.mysql.cj.jdbc.Driver");
        } catch (ClassNotFoundException e) {
            e.printStackTrace();
        }
    }

    public static synchronized ConexionBD getInstance() {
        if (instancia == null) {
            instancia = new ConexionBD();
        }
        return instancia;
    }

    public Connection getConnection() throws SQLException {
        return DriverManager.getConnection(URL, USER, PASS);
    }
}

package edu.usal.dao.interfaces;

import java.sql.Connection;
import java.sql.SQLException;
import java.util.List;
import edu.usal.domain.Persona;

public interface PersonaDAO {
    List<Persona> listarTodas(Connection con) throws SQLException;
    void crear(Persona persona, Connection con) throws SQLException;
    void actualizarPorDni(Persona persona, Connection con) throws SQLException;
    void eliminarPorDni(String dni, Connection con) throws SQLException;
}
PersonaDAOImpl.java
package edu.usal.dao.impl;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;

import edu.usal.dao.interfaces.PersonaDAO;
import edu.usal.domain.Persona;

public class PersonaDAOImpl implements PersonaDAO {

    @Override
    public List<Persona> listarTodas(Connection con) throws SQLException {
        List<Persona> lista = new ArrayList<>();
        String sql = "SELECT id, dni, nombre, apellido, edad FROM Personas";

        try (PreparedStatement ps = con.prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
            
            while (rs.next()) {
                Persona p = new Persona(
                    rs.getInt("id"),
                    rs.getString("dni"),
                    rs.getString("nombre"),
                    rs.getString("apellido"),
                    rs.getInt("edad")
                );
                lista.add(p);
            }
        }
        return lista;
    }

    @Override
    public void crear(Persona p, Connection con) throws SQLException {
        String sql = "INSERT INTO Personas (dni, nombre, apellido, edad) VALUES (?, ?, ?, ?)";
        try (PreparedStatement ps = con.prepareStatement(sql)) {
            ps.setString(1, p.getDni());
            ps.setString(2, p.getNombre());
            ps.setString(3, p.getApellido());
            ps.setInt(4, p.getEdad());
            ps.executeUpdate();
        }
    }

    @Override
    public void actualizarPorDni(Persona p, Connection con) throws SQLException {
        String sql = "UPDATE Personas SET nombre = ?, apellido = ?, edad = ? WHERE dni = ?";
        try (PreparedStatement ps = con.prepareStatement(sql)) {
            ps.setString(1, p.getNombre());
            ps.setString(2, p.getApellido());
            ps.setInt(3, p.getEdad());
            ps.setString(4, p.getDni());
            ps.executeUpdate();
        }
    }

    @Override
    public void eliminarPorDni(String dni, Connection con) throws SQLException {
        String sql = "DELETE FROM Personas WHERE dni = ?";
        try (PreparedStatement ps = con.prepareStatement(sql)) {
            ps.setString(1, dni);
            ps.executeUpdate();
        }
    }
}
5. Capa de Servicio (PersonaService.java)
Maneja la lógica de negocio y gestiona las transacciones JDBC (commit / rollback)
.
package edu.usal.service;

import java.sql.Connection;
import java.sql.SQLException;
import java.util.List;

import edu.usal.dao.impl.PersonaDAOImpl;
import edu.usal.dao.interfaces.PersonaDAO;
import edu.usal.domain.Persona;
import edu.usal.util.ConexionBD;

public class PersonaService {
    private PersonaDAO personaDAO = new PersonaDAOImpl();

    public List<Persona> listarPersonas() throws Exception {
        try (Connection con = ConexionBD.getInstance().getConnection()) {
            return personaDAO.listarTodas(con);
        }
    }

    public void crearPersona(Persona persona) throws Exception {
        try (Connection con = ConexionBD.getInstance().getConnection()) {
            con.setAutoCommit(false);
            try {
                personaDAO.crear(persona, con);
                con.commit();
            } catch (SQLException e) {
                con.rollback();
                throw e;
            }
        }
    }

    public void actualizarPersona(Persona persona) throws Exception {
        try (Connection con = ConexionBD.getInstance().getConnection()) {
            con.setAutoCommit(false);
            try {
                personaDAO.actualizarPorDni(persona, con);
                con.commit();
            } catch (SQLException e) {
                con.rollback();
                throw e;
            }
        }
    }

    public void eliminarPersona(String dni) throws Exception {
        try (Connection con = ConexionBD.getInstance().getConnection()) {
            con.setAutoCommit(false);
            try {
                personaDAO.eliminarPorDni(dni, con);
                con.commit();
            } catch (SQLException e) {
                con.rollback();
                throw e;
            }
        }
    }
}
6. Servlet Controlador (PersonaServlet.java)
Recibe las peticiones HTTP (crear, actualizar, eliminar o listar), delega en la capa de servicio y redirige al JSP
.
package edu.usal.controller;

import java.io.IOException;
import java.util.List;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

import edu.usal.domain.Persona;
import edu.usal.service.PersonaService;

@WebServlet("/PersonaServlet")
public class PersonaServlet extends HttpServlet {
    private static final long serialVersionUID = 1L;
    private PersonaService personaService = new PersonaService();

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response) 
            throws ServletException, IOException {
        try {
            List<Persona> lista = personaService.listarPersonas();
            request.setAttribute("listaPersonas", lista);
            request.getRequestDispatcher("index.jsp").forward(request, response);
        } catch (Exception e) {
            request.setAttribute("error", "Error al listar: " + e.getMessage());
            request.getRequestDispatcher("index.jsp").forward(request, response);
        }
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response) 
            throws ServletException, IOException {
        
        String accion = request.getParameter("accion");

        try {
            if ("crear".equals(accion)) {
                String dni = request.getParameter("dni");
                String nombre = request.getParameter("nombre");
                String apellido = request.getParameter("apellido");
                int edad = Integer.parseInt(request.getParameter("edad"));

                Persona p = new Persona(dni, nombre, apellido, edad);
                personaService.crearPersona(p);

            } else if ("actualizar".equals(accion)) {
                String dni = request.getParameter("dni");
                String nombre = request.getParameter("nombre");
                String apellido = request.getParameter("apellido");
                int edad = Integer.parseInt(request.getParameter("edad"));

                Persona p = new Persona(dni, nombre, apellido, edad);
                personaService.actualizarPersona(p);

            } else if ("eliminar".equals(accion)) {
                String dni = request.getParameter("dni");
                personaService.eliminarPersona(dni);
            }
        } catch (Exception e) {
            request.setAttribute("error", "Ocurrió un error: " + e.getMessage());
        }

        // Luego de cada operación POST, refrescamos la lista ejecutando doGet
        doGet(request, response);
    }
}
